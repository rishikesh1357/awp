const express = require("express");

app.use(express.json());

let students = [
    { id: 1, name: "Rahul", age: 20 },
    { id: 2, name: "Priya", age: 21 }
];

// CREATE - Add a student
app.post("/students", (req, res) => {
    students.push(req.body);
    res.send("Student added successfully");
});

// READ - Get all students
app.get("/students", (req, res) => {
    res.json(students);
});

// UPDATE - Update a student
app.put("/students/:id", (req, res) => {
    const id = Number(req.params.id);
    const student = students.find(s => s.id === id);

    if (!student) {
        return res.status(404).send("Student not found");

    student.name = req.body.name;
    student.age = req.body.age;

    res.send("Student updated successfully");
});

app.delete("/students/:id", (req, res) => {
    const id = Number(req.params.id);
    students = students.filter(s => s.id !== id);

    res.send("Student deleted successfully");
});

    console.log("Server running on port 3000");

const express = require("express");
const cors = require("cors");

const app = express();

// Middleware
app.use(cors());
app.use(express.json());

// Student data
let students = [
    { id: 1, name: "Rahul", age: 20 },
    { id: 2, name: "Priya", age: 21 }
];

// CREATE - Add a student
app.post("/students", (req, res) => {
    const student = req.body;
    students.push(student);

    res.status(201).json({
        message: "Student added successfully",
        student: student
    });
});

// READ - Get all students
app.get("/students", (req, res) => {
    res.json(students);
});

// READ - Get one student by ID
app.get("/students/:id", (req, res) => {
    const id = Number(req.params.id);

    const student = students.find(s => s.id === id);

    if (!student) {
        return res.status(404).send("Student not found");
    }

    res.json(student);
});

// UPDATE - Update a student
app.put("/students/:id", (req, res) => {
    const id = Number(req.params.id);

    const student = students.find(s => s.id === id);

    if (!student) {
        return res.status(404).send("Student not found");
    }

    student.name = req.body.name;
    student.age = req.body.age;

    res.json({
        message: "Student updated successfully",
        student: student
    });
});

// DELETE - Delete a student
app.delete("/students/:id", (req, res) => {
    const id = Number(req.params.id);

    const exists = students.some(s => s.id === id);

    if (!exists) {
        return res.status(404).send("Student not found");
    }

    students = students.filter(s => s.id !== id);

    res.send("Student deleted successfully");
});

// Start server
app.listen(3000, () => {
    console.log("Server running at http://localhost:3000");
});

        const url = "http://localhost:3000/students";

        // CREATE
        async function addStudent() {
            const student = {
                id: Number(document.getElementById("id").value),
                name: document.getElementById("name").value,
                age: Number(document.getElementById("age").value)
            };

            await fetch(url, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify(student)
            });

            getStudents();
        }

        // READ
        async function getStudents() {
            const response = await fetch(url);
            const students = await response.json();

            const list = document.getElementById("list");
            list.innerHTML = "";

            students.forEach(student => {
                const li = document.createElement("li");
                li.textContent =
                    student.id + " - " +
                    student.name + " - " +
                    student.age;
                list.appendChild(li);
            });
        }

        // UPDATE
        async function updateStudent() {
            const id = document.getElementById("id").value;

            const student = {
                name: document.getElementById("name").value,
                age: Number(document.getElementById("age").value)
            };

            await fetch(url + "/" + id, {
                method: "PUT",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify(student)
            });

            getStudents();
        }

        // DELETE
        async function deleteStudent() {
            const id = document.getElementById("id").value;

            await fetch(url + "/" + id, {
                method: "DELETE"
            });

            getStudents();
        }

        // Load students when page opens
        getStudents();

6) For the above application create authorized end points using JWT (JSON Web Token).

const express = require("express");
const jwt = require("jsonwebtoken");

const app = express();
app.use(express.json());
const secret = "admin123";

let students = [
    { id: 1, name: "rishi" },
    { id: 2, name: "anu" }
];

app.post("/login", (req, res) => {
    const token = jwt.sign(
        { user: req.body.user }, secret
    );
    res.json({ token: token });
});

function auth(req, res, next) {
    try {
        jwt.verify(req.headers.authorization.split(" ")[1], secret);
        next();
    } catch {
        res.status(401).send("invalid");
    }
}

app.get("/students", auth, (req, res) => {
    res.json(students);
});

app.put("/students/:id", auth, (req, res) => {
    const id = Number(req.params.id);
    const student = students.find(s => s.id === id);

    if (student) {
        student.name = req.body.name;
        res.send("successfully updated");
    } else {
        res.status(404).send("Student not found");
    }
});

app.delete("/students/:id", auth, (req, res) => {
    const id = Number(req.params.id);
    students = students.filter(s => s.id !== id);
    res.send("successfully deleted");
});

app.listen(3000);

13) Create a react application to implement the calculator application by creating UI and providing the functionality using events. Also make it look good by using styles.
    (React Styling: How to use the Styles in React)

import { useState } from "react";
import './App.css'
function App() {
  const [a, setA] = useState("");
  const [b, setB] = useState("");
  const [result, setResult] = useState("");

  function calculate(op) {
    let x = Number(a);
    let y = Number(b);

    if (op === "+")
      setResult(x + y);
    else if (op === "-")
      setResult(x - y);
    else if (op === "*")
      setResult(x * y);
    else if (op === "/") {
      if (y === 0)
        setResult("Not possible");
      else
        setResult(x / y);
    }
  }

  return (
    <div className="calculator">
      <h2>Calculator</h2>

      <input type="number"
        onChange={(e) => setA(e.target.value)} />

      <input type="number"
        onChange={(e) => setB(e.target.value)} />

      <div>
        <button onClick={() => calculate("+")}>+</button>
        <button onClick={() => calculate("-")}>-</button>
        <button onClick={() => calculate("*")}>*</button>
        <button onClick={() => calculate("/")}>/</button>
      </div>

      <h2>{result}</h2>
    </div>
  );
}

export default App;

14) Create a react application to create a registration page using forms as controlled components.
    (Working with the React Forms)


import { useState } from "react";

function App() {
  const [name, setName] = useState("");
  const [pass, setPass] = useState("");
  const [result, setResult] = useState(false);

  function display(e) {
    e.preventDefault();
    setResult(true);
  }

  return (
    <div>
      <form onSubmit={display}>
        <label>Name</label>
        <br />
        <input
          type="text"
          value={name}
          onChange={(e) => setName(e.target.value)}
          required
        />

        <br /><br />

        <label>Password</label>
        <br />
        <input
          type="password"
          value={pass}
          onChange={(e) => setPass(e.target.value)}
          required
        />

        <br /><br />
        <button type="submit">Submit</button>
      </form>

      {result && (
        <div>
          <h3>Registration Successful!</h3>
          <p>Name: {name}</p>
        </div>
      )}
    </div>
  );
}

export default App;

16) Create a simple e-commerce storefront for a bookstore which consists of four screens:
    Home — The storefront with a book list
    Product page — A separate product page
    Cart — A web page showing the quantities and titles selected by the user.
    Checkout — A print-ready invoice with the list of books


import React, { useState } from 'react';

function App() {
    const [page, setPage] = useState(1);
    const [cart, setCart] = useState([]);
    const [book, setBook] = useState(null);

    const books = [
        { id: 1, title: "Python Programming", price: 450 },
        { id: 2, title: "Machine Learning", price: 550 },
        { id: 3, title: "Web Development", price: 400 },
        { id: 4, title: "Data Science", price: 600 }
    ];

    return (
        <div>
            <h1>Book Store</h1>

            <button onClick={() => setPage(1)}>Home</button>
            <button onClick={() => setPage(3)}>Cart</button>

            {page === 1 && (
                <>
                    <h2>Home Page</h2>

                    {books.map((b) => (
                        <p key={b.id}>
                            {b.title} - ₹{b.price}

                            <button onClick={() => {
                                setBook(b);
                                setPage(2);
                            }}>
                                View Product
                            </button>
                        </p>
                    ))}
                </>
            )}

            {page === 2 && (
                <>
                    <h2>Product Details</h2>
                    <h3>{book.title}</h3>
                    <p>Price: ₹{book.price}</p>

                    <button onClick={() => setCart([...cart, book])}>
                        Add to Cart
                    </button>
                </>
            )}

            {page === 3 && (
                <>
                    <h2>Shopping Cart</h2>

                    {cart.map((b, i) => (
                        <p key={i}>
                            {b.title} - ₹{b.price}
                        </p>
                    ))}

                    <h3>
                        Total: ₹{cart.reduce((sum, b) => sum + b.price, 0)}
                    </h3>

                    <button onClick={() => setPage(4)}>
                        Checkout
                    </button>
                </>
            )}

            {page === 4 && (
                <>
                    <h2>Checkout</h2>
                    <h3>Order placed successfully!</h3>

                    <button onClick={() => window.print()}>
                        Print Invoice
                    </button>
                </>
            )}
        </div>
    );
}

export default App;
